CREATE TABLE `VENDORS` (
`VND_CODE` VARCHAR(6) NOT NULL COMMENT 'Primary Key รหัสอ้างอิงคู่ค้า',
`VND_NAME` VARCHAR(150) NOT NULL COMMENT 'ชื่อย่อ หรือชื่อทางการค้าของ Vendor',
`VND_LONGNAME` VARCHAR(255) NULL DEFAULT NULL COMMENT 'ชื่อเต็มบริษัท (Legal Name)',
`VND_TYPE` VARCHAR(50) NULL DEFAULT NULL COMMENT 'ประเภทของ Vendor/ผู้ผลิต',
`VND_STATUS` TINYINT NOT NULL DEFAULT 1 COMMENT '1=Active, 2=Inactive, 4=Pending Approval',
`VND_REGDATE` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'วันที่ลงทะเบียนคู่ค้า',
`VND_LASTUPDATE` DATETIME NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'วันที่แก้ไขล่าสุด',
`VND_USERUPDATE` VARCHAR(50) NULL DEFAULT NULL COMMENT 'รหัสพนักงาน/ผู้ที่ทำการอัปเดตข้อมูล',
PRIMARY KEY (`VND_CODE`),
KEY `IX_VENDORS_STATUS` (`VND_STATUS`),
KEY `IX_VENDORS_NAME` (`VND_NAME`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='ตารางเก็บข้อมูลบริษัทคู่ค้า/ซัพพลายเออร์';

-- 2. ตารางรหัสคู่ค้า (VENDORS_CODE) - รองรับ 1 Vendor หลายรหัส
CREATE TABLE `VENDORS_GROUP` (
`CODE_ID` INT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT 'Primary Key รหัสรายการ',
`CODE_NUM` VARCHAR(50) NOT NULL COMMENT 'รหัส Vendor Code (เช่น รหัสขึ้นต้น 8xxxx), FK อ้างอิง VENDORS.VND_CODE',
`VENDOR_GROUP` INT NOT NULL,
`CODE_STATUS` TINYINT NOT NULL DEFAULT 1 COMMENT '1=Active, 0=Inactive',
`CODE_REGDATE` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'วันที่ลงทะเบียนรหัส',
PRIMARY KEY (`CODE_ID`),
UNIQUE KEY `UQ_CODE_NUM` (`CODE_NUM`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='ตารางจัดเก็บรหัส Vendor Code ของคู่ค้า';

-- 3. ตารางที่อยู่ของคู่ค้า (VENDORS_ADDRESS)
CREATE TABLE `VENDORS_ADDRESS` (
`ADDR_ID` INT UNSIGNED NOT NULL AUTO_INCREMENT,
`VENDOR_ID` VARCHAR(5) NOT NULL,
`ADDR_LINE1` VARCHAR(255) NULL DEFAULT NULL,
`ADDR_LINE2` VARCHAR(255) NULL DEFAULT NULL,
`ADDR_LINE3` VARCHAR(255) NULL DEFAULT NULL,
`ADDR_CITY` VARCHAR(100) NULL DEFAULT NULL,
`ADDR_COUNTRY` INT NULL DEFAULT NULL,
`ADDR_ZIPCODE` VARCHAR(20) NULL DEFAULT NULL,
`ADDR_STATUS` TINYINT NOT NULL DEFAULT 1,
PRIMARY KEY (`ADDR_ID`),
KEY `IX_VENDORS_ADDR_VENDOR` (`VENDOR_ID`),
CONSTRAINT `FK_VENDORS_ADDR_VENDOR`
FOREIGN KEY (`VENDOR_ID`) REFERENCES `VENDORS` (`VND_CODE`)
ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='ตารางที่อยู่บริษัทคู่ค้า';
